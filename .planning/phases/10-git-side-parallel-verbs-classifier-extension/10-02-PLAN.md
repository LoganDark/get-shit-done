performGitParallelDispatch(opts):
  // D-05 + Pitfall 4 (Pitfall 5 in PITFALLS.md): plain sync for-loop. The
  // MANDATORY inline comment at this loop's header cites PITFALLS Pitfall 5
  // and notes spawnSync as the structural protection (verbatim per CONTEXT D-05).
  let phaseTag = String(phaseNumber).padStart(2, '0')
  let slots = []
  for (let i = 0; i < plan.length; i++):
    let item = plan[i]
    let idx = i + 1
    validateAgentId(item.agentId)                                      // inline regex
    let workspaceName = `phase-${phaseTag}-subagent-${idx}`              // mirror octopus.ts:30
    let agentBookmark = `worktree-agent-${item.agentId}`                 // matches refname regex
    let wsPath = item.workspacePath ?? <default path under .git/gsd-workspaces>
    opts.vcs.workspace.add({ path: wsPath, baseRef: item.baseRef, name: agentBookmark })
    // After workspace.add, the agent's branch tip = baseRef. Capture for handle.
    let headSha = (vcsExec(mainRepoRoot, 'git', ['rev-parse', agentBookmark])).stdout.trim()
    slots.push({ agentId, planId, workspacePath: wsPath, workspaceName, headSha })

  // Manifest write — VCS-19 schema (mirror jj/parallel.ts:252-265, change backend literal).
  let manifestDir = mkdtempSync(join(tmpdir(), 'gsd-wave-manifest-'))
  let manifestPath = join(manifestDir, 'wave-worktree-manifest.json')
  let manifestBody = { worktrees: slots.map(s => ({
    agent_id: s.agentId,
    plan_id: s.planId,
    backend: 'git' as const,                                            // ← CHANGE FROM 'jj'
    worktree_path: s.workspacePath,
    branch: `worktree-agent-${s.agentId}`,
    expected_base: s.headSha,
    main_bookmark: opts.mainBookmark,
  })) }
  writeFileSync(manifestPath, JSON.stringify(manifestBody, null, 2), 'utf-8')

  // Frozen pure-JSON return — verbatim mirror of jj/parallel.ts:270-289.
  return Object.freeze({
    phaseRoot: derivePhaseRoot(mainRepoRoot, opts.phaseNumber),
    phaseNumber: opts.phaseNumber,
    mainBookmark: opts.mainBookmark,
    manifest: manifestPath,
    workspaces: Object.freeze(slots.map(s => Object.freeze({
      name: s.workspaceName,
      path: s.workspacePath,
      baseRev: s.headSha,
      agentId: s.agentId,
      baselineOpId: undefined as string | undefined,
    }))),
  }) satisfies ParallelDispatchHandle

performGitParallelFanIn(mainRepoRoot, handle, results):
  let phaseTag = String(handle.phaseNumber).padStart(2, '0')
  let merged: string[] = []
  let conflicted = false
  let conflictedPaths: string[] = []
  let incompleteQueued = 0
  let failedReaped: string[] = []
  let surplusBookmarks: string[] = []

  // STEP 1 — fan-in loop (D-01..D-04, D-08).
  for (let ws of handle.workspaces):
    let agentBookmark = `worktree-agent-${ws.agentId}`

    // D-03 stateless re-call probe. Pass branch NAME (not SHA) — Open Q3
    // recommendation: treat any non-zero exit as "skip" (covers deleted-branch
    // case where exit=128 ALSO means "this workspace already merged").
    let probe = vcsExec(mainRepoRoot, 'git', ['merge-base', '--is-ancestor', agentBookmark, 'HEAD'])
    if (probe.exitCode === 0):
      continue   // already merged on a previous call — skip per Pitfall 3 (do NOT push to merged[])
    // probe.exitCode !== 0 covers BOTH "not ancestor" (=1) AND "branch missing" (=128) — both = skip-or-process
    // For exitCode 128 (no such ref) → definitively skip (the branch is gone, must have been deleted on prior call).
    if (probe.exitCode === 128 || probe.stderr.includes('Not a valid commit name')):
      continue

    // D-01 + D-02 + D-08: 2-parent merge via the existing primitive.
    // The DI seam is the curried `vcs.workspace` captured by the wire-in
    // in Plan 10.03. See <interfaces> RECOMMENDATION above for exact shape.
    let mergeResult = vcs.workspace.merge({
      branch: expr.bookmark(agentBookmark),   // or expr.rev(...) per locked types
      message: `phase ${phaseTag} merge: ${ws.agentId}`,
      ff: false,
      mainBookmark: handle.mainBookmark,
      agentBookmark,                          // for in-merge delete attempt
    })

    if (!mergeResult.ok && mergeResult.conflicted):
      // D-02 halt + D-08 producer.
      let diff = vcsExec(mainRepoRoot, 'git', ['diff', '--name-only', '--diff-filter=U'])
      conflictedPaths = diff.exitCode === 0
        ? diff.stdout.split('\n').map(s => s.trim()).filter(s => s.length > 0)
        : []
      conflicted = true
      let tipSha = vcsExec(mainRepoRoot, 'git', ['rev-parse', agentBookmark]).stdout.trim()
      appendIncomplete(handle.phaseRoot, {
        subagentName: ws.name,
        changeIdShort: tipSha.slice(0, 12),
        workspacePath: ws.path,
        reason: 'merge-in-tree-conflict',
      })
      incompleteQueued += 1
      break   // D-02: halt the loop, tree left wedged with MERGE_HEAD set

    if (mergeResult.ok && mergeResult.changeId):
      // D-04 per-success cleanup.
      merged.push(mergeResult.changeId.slice(0, 12))
      try:
        vcs.workspace.remove(ws.path)   // non-force per D-07
      catch (err):
        // D-07 + Pitfall 5: non-force refuse on unexpectedly-dirty WC IS the feature.
        // Record as surplus; DO NOT re-throw.
        surplusBookmarks.push(agentBookmark)
      // Pitfall 2: bookmark may survive workspace.merge's in-step delete on git.
      // Defer to surplusBookmarks audit below (incremental, per CONTEXT recommendation).

  // STEP 2 — crashed-agent classifier (D-06 + D-07). Only runs on results whose
  // exitCode !== 0. Skips workspaces that already merged successfully above
  // (they're not in `results` because the orchestrator didn't surface them as failures).
  for (let result of results):
    if (result.exitCode === 0) continue
    let ws = handle.workspaces.find(w => w.agentId === result.agentId)
    if (!ws) continue   // unknown agentId — shouldn't happen but skip defensively
    let agentBookmark = `worktree-agent-${ws.agentId}`
    let tipResult = vcsExec(mainRepoRoot, 'git', ['rev-parse', agentBookmark])
    if (tipResult.exitCode !== 0):
      // Branch missing — bookmark already cleaned up (e.g., agent crashed AFTER successful merge).
      failedReaped.push(ws.name)
      continue
    let tipSha = tipResult.stdout.trim()
    // Probe the worktree directory's WC dirtiness with `git status --porcelain`.
    // The worktree may or may not still exist (D-07 may have refused removal earlier).
    let statusResult = vcsExec(ws.path, 'git', ['status', '--porcelain'])
    let cleanTree = statusResult.exitCode === 0 && statusResult.stdout.trim() === ''

    // D-06: branch tip == baseRev AND clean → no queue write (abandoned-style).
    if (tipSha === ws.baseRev && cleanTree):
      failedReaped.push(ws.name)
      continue

    // Otherwise (committed work OR dirty tree OR both) → queue entry.
    // CONTEXT Claude's Discretion + 10-RESEARCH Open Q2 recommendation:
    // queue as 'crashed-with-uncommitted-work' even when committed-but-clean
    // (the committed work IS uncommitted-from-main-perspective).
    appendIncomplete(handle.phaseRoot, {
      subagentName: ws.name,
      changeIdShort: tipSha.slice(0, 12),
      workspacePath: ws.path,
      reason: 'crashed-with-uncommitted-work',
    })
    incompleteQueued += 1
    // D-07: try non-force cleanup; surface refusal as surplus, do not throw.
    try:
      vcs.workspace.remove(ws.path)
    catch (err):
      // EXPECTED for dirty trees — worktree survives on disk for human inspection.
      // Do NOT push to surplusBookmarks here; the branch surfaces via the audit below
      // OR the workspacePath in the queue entry is the inspection handle.
    failedReaped.push(ws.name)

  // STEP 3 — surplus-bookmark audit (CONTEXT recommendation: incremental + final).
  // List remaining `worktree-agent-*` refs after the loops; any that the in-loop
  // accumulation already added are deduped; any new ones surface defensively.
  let listResult = vcsExec(mainRepoRoot, 'git', ['for-each-ref', '--format=%(refname:short)', 'refs/heads/worktree-agent-*'])
  if (listResult.exitCode === 0):
    let alive = listResult.stdout.split('\n').map(s => s.trim()).filter(s => s.length > 0)
    for (let bm of alive):
      if (!surplusBookmarks.includes(bm)):
        surplusBookmarks.push(bm)

  // Frozen pure-JSON return — verbatim mirror of jj/parallel.ts:535-542.
  return Object.freeze({
    merged: Object.freeze(merged.slice()) as readonly string[],
    conflicted,
    conflictedPaths: Object.freeze(conflictedPaths.slice()) as readonly string[],
    incompleteQueued,
    failedReaped: Object.freeze(failedReaped.slice()) as readonly string[],
    surplusBookmarks: Object.freeze(surplusBookmarks.slice()) as readonly string[],
  }) satisfies FanInResult
</algorithm_pseudocode>
